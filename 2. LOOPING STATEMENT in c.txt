2. LOOPING STATEMENT PROGRAMS (for, while, do-while)

1️⃣ Print 1 to 10

#include <stdio.h>
int main() {
    for (int i = 1; i <= 10; i++)
        printf("%d ", i);
    return 0;
}

2️⃣ Sum of first n natural numbers

#include <stdio.h>
int main() {
    int n, sum = 0;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++)
        sum += i;
    printf("%d", sum);
    return 0;
}

3️⃣ Factorial of a number

#include <stdio.h>
int main() {
    int n, fact = 1;
    scanf("%d", &n);
    for (int i = 1; i <= n; i++)
        fact *= i;
    printf("%d", fact);
    return 0;
}

4️⃣ Multiplication table

#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    for (int i = 1; i <= 10; i++)
        printf("%d x %d = %d\n", n, i, n * i);
    return 0;
}

5️⃣ Reverse of a number

#include <stdio.h>
int main() {
    int n, rev = 0;
    scanf("%d", &n);
    while (n != 0) {
        rev = rev * 10 + n % 10;
        n /= 10;
    }
    printf("%d", rev);
    return 0;
}

6️⃣ Palindrome number

#include <stdio.h>
int main() {
    int n, temp, rev = 0;
    scanf("%d", &n);
    temp = n;
    while (temp != 0) {
        rev = rev * 10 + temp % 10;
        temp /= 10;
    }
    if (rev == n) printf("Palindrome");
    else printf("Not palindrome");
    return 0;
}

7️⃣ Fibonacci series

#include <stdio.h>
int main() {
    int n, a = 0, b = 1, c;
    scanf("%d", &n);
    printf("%d %d ", a, b);
    for (int i = 2; i < n; i++) {
        c = a + b;
        printf("%d ", c);
        a = b; b = c;
    }
    return 0;
}

8️⃣ Count digits

#include <stdio.h>
int main() {
    int n, count = 0;
    scanf("%d", &n);
    while (n != 0) {
        count++;
        n /= 10;
    }
    printf("%d", count);
    return 0;
}


9️⃣ Sum of digits

#include <stdio.h>
int main() {
    int n, sum = 0;
    scanf("%d", &n);
    while (n != 0) {
        sum += n % 10;
        n /= 10;
    }
    printf("%d", sum);
    return 0;
}


🔟 Armstrong number

#include <stdio.h>
#include <math.h>
int main() {
    int n, temp, rem, sum = 0, digits = 0;
    scanf("%d", &n);
    temp = n;
    while (temp != 0) { digits++; temp /= 10; }
    temp = n;
    while (temp != 0) {
        rem = temp % 10;
        sum += pow(rem, digits);
        temp /= 10;
    }
    if (sum == n) printf("Armstrong");
    else printf("Not Armstrong");
    return 0;
}
