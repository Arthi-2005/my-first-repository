1️⃣ Check even or odd

#include <stdio.h>
int main() {
    int n;
    printf("Enter a number: ");
    scanf("%d", &n);
    if (n % 2 == 0)
        printf("Even");
    else
        printf("Odd");
    return 0;
}


2️⃣ Find largest of two numbers

#include <stdio.h>
int main() {
    int a, b;
    scanf("%d%d", &a, &b);
    if (a > b)
        printf("A is largest");
    else
        printf("B is largest");
    return 0;
}



3️⃣ Find largest of three numbers

#include <stdio.h>
int main() {
    int a, b, c;
    scanf("%d%d%d", &a, &b, &c);
    if (a > b && a > c)
        printf("A is largest");
    else if (b > c)
        printf("B is largest");
    else
        printf("C is largest");
    return 0;
}



4️⃣ Check positive, negative, or zero

#include <stdio.h>
int main() {
    int n;

    if (n > 0)
        printf("Positive");

}


5️⃣ Check leap year

#include <stdio.h>
int main() {
    int y;
    scanf("%d", &y);
    if ((y % 4 == 0 && y % 100 != 0) || y % 400 == 0)
        printf("Leap year");
    else
        printf("Not leap year");
    return 0;
}


6️⃣ Grade of marks

#include <stdio.h>
int main() {
    int m;
    scanf("%d", &m);
    if (m >= 90) printf("Grade A");
    else if (m >= 75) printf("Grade B");
    else if (m >= 50) printf("Grade C");
    else printf("Fail");
    return 0;
}


7️⃣ Vowel or consonant using switch

#include <stdio.h>
int main() {
    char ch;
    scanf(" %c", &ch);
    switch (ch) {
        case 'a': case 'e': case 'i': case 'o': case 'u':
        case 'A': case 'E': case 'I': case 'O': case 'U':
            printf("Vowel");
            break;
        default:
            printf("Consonant");
    }
    return 0;
}


8️⃣ Simple calculator using switch

#include <stdio.h>
int main() {
    int a, b;
    char op;
    scanf("%d %d %c", &a, &b, &op);
    switch (op) {
        case '+': printf("%d", a + b); break;
        case '-': printf("%d", a - b); break;
        case '*': printf("%d", a * b); break;
        case '/': printf("%d", a / b); break;
        default: printf("Invalid");
    }
    return 0;
}


9️⃣ Check odd or even using switch

#include <stdio.h>
int main() {
    int n;
    scanf("%d", &n);
    switch (n % 2) {
        case 0: printf("Even"); break;
        default: printf("Odd");
    }
    return 0;
}
